async function save(d: any) {
  const { mergeRangeWorkspace } = await import(REPO + "workspace-merge.ts");
  let r = await put(d.local, d.v);
  if (r.status === 409 && r.body.code === "stale_workspace") {
    const server = clone(strip(stored.workspace));
    const m = mergeRangeWorkspace(d.base, d.local, server);
    d.conflicts.push(...m.conflicts); d.local = m.merged; d.base = server; d.v = stored.updated_at;
    r = await put(d.local, d.v);
  }
  if (r.status === 200) { d.base = clone(d.local); d.v = r.body.updatedAt; }
  return r;
}

test("P1 same officer/drill/run scored fresh on two devices -> two results stored, no conflict", async () => {
  const [one, two] = await start();
  one.local.results = [res("result-1000-0", "officer-a", 46)];
  two.local.results = [res("result-1003-0", "officer-a", 31)];
  assert.equal((await save(one)).status, 200);
  assert.equal((await save(two)).status, 200);
  assert.equal(two.conflicts.length, 0);
  const forA = stored.workspace.results.filter((r: any) => r.officerId === "officer-a" && r.runNumber === 1);
  assert.deepEqual(forA.map((r: any) => r.score), [46, 31]);
});

test("P2 other device finalizes the day; recovered save is refused (409 w/o stale code) and every later save fails", async () => {
  const [lead, line] = await start();
  lead.local.rangeDays = [{ ...lead.local.rangeDays[0], status: "Completed" }];
  assert.equal((await save(lead)).status, 200);
  ctx.permissions = ["score_range_days"];
  line.local.results = [res("result-2000-0", "officer-a", 44)];
  const r = await save(line);
  assert.equal(r.status, 409);
  assert.equal(r.body.code, undefined); // client throws generic Error -> no recovery, no conflict
  assert.equal(line.conflicts.length, 0);
  // any later unrelated edit is still refused because the whole workspace carries the replayed score
  line.local.rangeRoster = line.local.rangeRoster.map((e: any) => ({ ...e, attended: true }));
  assert.equal((await save(line)).status, 409);
});

test("P4 base/local pair mismatch (2 tabs share localStorage) silently reverts a saved correction and deletes a roster entry", async () => {
  const { mergeRangeWorkspace } = await import(REPO + "workspace-merge.ts");
  const p = plan(); p.results = [res("result-1", "officer-a", 40)];
  const [tab1, tab2] = await start(p);
  // tab1 corrects the score and adds a roster entry; saved -> sync state {base: tab1 state, v2}
  tab1.local.results = [res("result-1", "officer-a", 46)];
  tab1.local.rangeRoster = [...tab1.local.rangeRoster, { id: "roster-new", rangeDayId: "day-1", officerId: "officer-b" }];
  assert.equal((await save(tab1)).status, 200);
  const syncStateFromTab1 = { base: clone(tab1.base), v: tab1.v };
  // tab2 (still showing v1) edits a day title -> writes the shared local-copy key (no correction, no roster-new)
  const localCopyFromTab2 = clone(tab2.local); localCopyFromTab2.rangeDays[0].title = "edited in tab 2";
  // reload (either tab): loadWorkspace merges syncState.base with the local copy and the server
  const server = clone(strip(stored.workspace));
  const m = mergeRangeWorkspace(syncStateFromTab1.base, localCopyFromTab2, server);
  assert.equal(m.conflicts.length, 0);
  assert.equal(m.merged.results[0].score, 40);                                   // correction reverted
  assert.ok(!m.merged.rangeRoster.some((e: any) => e.id === "roster-new"));       // entry deleted
  assert.equal((await put(m.merged, stored.updated_at)).status, 200);             // and the server accepts it
  assert.equal(stored.workspace.results[0].score, 40);
});

test("P7 repeated conflict on the same record is listed twice; resolving the older one applies a superseded value", async () => {
  const { mergeRangeWorkspace, applyConflictChoice } = await import(REPO + "workspace-merge.ts");
  const W = (s: number) => ({ ...plan(), results: [res("r", "officer-a", s)] }) as any;
  let conflicts: any[] = [];
  let m = mergeRangeWorkspace(W(40), W(44), W(46)); conflicts.push(...m.conflicts); // first 409
  // user re-enters 48 for the same officer; other device saves 47; second 409
  m = mergeRangeWorkspace(W(46), W(48), W(47)); conflicts.push(...m.conflicts);
  assert.deepEqual(conflicts.map((c) => `${c.collection}:${c.id}`), ["results:r", "results:r"]); // duplicate React keys
  const ws = applyConflictChoice(m.merged, conflicts[0]);
  assert.equal(ws.results[0].score, 44); // the user's later 48 is not what gets applied
});
