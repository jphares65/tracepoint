type Device = { base: any; baseUpdatedAt: string | null; local: any; conflicts: any[]; canManage: boolean; refusalRecoveredFor?: string | null };

// Mirrors page.tsx recoverRangeWorkspace + serverAcceptance (preview against the fetched server copy).
async function recover(device: Device) {
  const { recoverWorkspace } = await import(`${REPO}/lib/range/workspace-device-copy.ts`);
  const { previewRangeWorkspaceMutation } = await import(`${REPO}/lib/range/workspace-mutation.ts`);
  const serverRaw = clone(stored!.workspace);
  const server = strip(serverRaw);
  const recovered = recoverWorkspace({
    base: device.base, local: device.local, server, conflicts: device.conflicts,
    accept: (workspace: any) => {
      const decision = previewRangeWorkspaceMutation({ existingWorkspace: serverRaw, nextWorkspace: workspace, canManage: device.canManage, canScore: true });
      return decision.ok ? { ok: true as const } : { ok: false as const, error: decision.error };
    },
  });
  device.conflicts = recovered.conflicts;
  device.local = recovered.workspace;
  device.base = server;
  device.baseUpdatedAt = stored!.updated_at;
}

// Mirrors saveRangeDayWorkspaceToServer: stale -> recover (+ the autosave that follows);
// other refusal -> recover once per server version (refusalRecoveredFor).
async function save(device: Device) {
  let response = await put(device.local, device.baseUpdatedAt);
  if (response.status === 409 && response.body.code === "stale_workspace") {
    await recover(device);
    response = await put(device.local, device.baseUpdatedAt);
  } else if ([400, 403, 409].includes(response.status) && device.refusalRecoveredFor !== device.baseUpdatedAt) {
    device.refusalRecoveredFor = device.baseUpdatedAt;
    await recover(device);
    response = await put(device.local, device.baseUpdatedAt);
  }
  if (response.status === 200) { device.base = clone(device.local); device.baseUpdatedAt = response.body.updatedAt; }
  return response;
}

async function startDay(initial: any = plan()) {

  return load;
}

  const load = await startDay();
  const lead = load(true);
  const line = load(false);
  // Line instructor marks officer A present and it saves (v2).
  as(["score_range_days"]);
  line.local.rangeRoster = line.local.rangeRoster.map((entry: any) => (entry.id === "roster-a" ? { ...entry, attended: true } : entry));
  assert.equal((await save(line)).status, 200);
  assert.equal(stored!.workspace.rangeRoster[0].attended, true);
  // Lead (still on v1) has added drill-3 locally (not yet saved) and removes drill-2.
  as(["manage_range_days"]);

  const removed = await removeDrill("day-1", "drill-2");
  assert.equal(removed.status, 200, "the DELETE route has no version check");
  // page.tsx handleRemoveDrillFromSelectedRangeDay: base := payload.updatedAt; drills := payload.workspace.rangeDayDrills.
  lead.baseUpdatedAt = removed.body.updatedAt;
  lead.local.rangeDayDrills = removed.body.workspace.rangeDayDrills;
  assert.ok(!lead.local.rangeDayDrills.some((drill: any) => drill.id === "drill-3"), "this device's unsaved drill is discarded");
  // The autosave triggered by setRangeDayDrills.
  const autosave = await save(lead);
  assert.equal(autosave.status, 200, "accepted: version check passes, no 409, no conflict");
  assert.equal(stored!.workspace.rangeRoster[0].attended, false, "the line instructor's saved attendance was silently reverted");

test("R2-M1: a conflict outlives a later edit; 'Keep other device's value' then writes the stale server snapshot over the instructor's newer saved score", async () => {
  const { applyConflictChoice } = await import(`${REPO}/lib/range/workspace-merge.ts`);
  const load = await startDay({ ...plan(), results: [{ id: "result-a", rangeDayId: "day-1", drillId: "drill-1", runNumber: 1, officerId: "officer-a", score: 40 }] });
  const one = load(false);
  const two = load(false);
  one.local.results = [{ ...one.local.results[0], score: 46 }];
  two.local.results = [{ ...two.local.results[0], score: 44 }];
  assert.equal((await save(one)).status, 200);
  assert.equal((await save(two)).status, 200);
  assert.deepEqual(two.conflicts.map((c: any) => [c.local.score, c.server.score]), [[44, 46]]);
  // The instructor on device two re-enters the score in the row (48); it saves.
  two.local.results = [{ ...two.local.results[0], score: 48 }];
  assert.equal((await save(two)).status, 200);
  assert.equal(stored!.workspace.results[0].score, 48);
  assert.equal(two.conflicts.length, 1, "the conflict is still listed (saves never reconcile)");
  // Clicking "Keep other device's value" (thinking it keeps what is now in use)…
  two.local = applyConflictChoice(two.local, two.conflicts[0], "server");
  two.conflicts = [];
  assert.equal((await save(two)).status, 200);
  assert.equal(stored!.workspace.results[0].score, 46, "…writes 46, a value no device currently holds, over the saved 48");

test("R2-M3: a refusal the client preview cannot foresee (agency/reference check) still wedges every save, survives a reload, and lists no conflict", async () => {
  const load = await startDay();
  // A lead planned day-2 with a roster and a practice score while sync was blocked; their
  // manage permission was then removed (now a scorer). The copy's base is v1.
  const device = load(false);
  device.local.rangeDays = [...device.local.rangeDays, { id: "day-2", departmentId: department, status: "Planned" }];
  device.local.rangeDayDrills = [...device.local.rangeDayDrills, { id: "drill-9", rangeDayId: "day-2", name: "Practice", maxScore: 50 }];
  device.local.rangeRoster = [...device.local.rangeRoster, { id: "roster-9", rangeDayId: "day-2", officerId: "officer-c", attended: true }];
  device.local.results = [{ ...fresh("officer-c", 45), rangeDayId: "day-2", drillId: "drill-9" }];
  // Load-time recovery (page.tsx loadWorkspace -> recoverWorkspace with serverAcceptance).
  await recover(device);
  assert.ok(device.conflicts.some((c: any) => c.collection === "rangeDays"), "the day is held back (planning)");
  assert.ok(!device.conflicts.some((c: any) => c.collection === "results"), "but its result is NOT held back: the preview has no reference check");
  // Unrelated work: officer A attendance.
  device.local.rangeRoster = device.local.rangeRoster.map((entry: any) => (entry.id === "roster-a" ? { ...entry, attended: true } : entry));
  const first = await save(device);
  assert.equal(first.status, 403);
  assert.match(first.body.error, /outside your agency/);
  // The one refusal recovery re-runs the same preview, so the retry is refused again; later saves are refused forever.
  const second = await save(device);
  assert.equal(second.status, 403);
  assert.equal(stored!.workspace.rangeRoster[0].attended, false, "the unrelated attendance never saves");
  assert.ok(!device.conflicts.some((c: any) => c.collection === "results"), "nothing for the instructor to discard");
  // Reload: the copy (base = server, workspace = local) is merged again with the same preview.
  device.refusalRecoveredFor = undefined;
  await recover(device);
  assert.equal((await save(device)).status, 403, "still wedged after reload");
});
